% The standard class, not KOMA-Script: the caption package of TeX Live 2025 does not recognize
% the current scrreprt and then attaches no number to \label (every figure reference broke).
\documentclass[a4paper, 11pt, oneside]{report}

% --- Page layout and headings --------------------------------------------------------------
\usepackage[a4paper, top=30mm, bottom=30mm, inner=30mm, outer=25mm]{geometry}
\usepackage[skip=0.5\baselineskip plus 2pt]{parskip}
\usepackage{titlesec}
% Chapters continue on the same page: the report is limited to 10 pages without the appendix.
\titleclass{\chapter}{straight}
\titleformat{\chapter}[hang]{\normalfont\LARGE\bfseries}{\thechapter\quad}{0pt}{}
\titlespacing*{\chapter}{0pt}{3.5ex plus 1ex minus .2ex}{2ex plus .2ex}
\usepackage[nottoc]{tocbibind} % lists of figures and tables in the table of contents

% --- Language and fonts --------------------------------------------------------------------
\usepackage{polyglossia}
\setdefaultlanguage[variant=american]{english}
\setotherlanguage{german} % statutory declaration

\usepackage{amsmath}
\usepackage{fontspec}
\usepackage{unicode-math}
\setmathfont{Latin Modern Math}
\usepackage{microtype}
\usepackage{csquotes}

% --- Floats, tables, units -----------------------------------------------------------------
\usepackage{graphicx}
\usepackage[labelfont=bf, font=small]{caption}
\usepackage{subcaption}
\usepackage{booktabs}
\usepackage{tabularx}
\usepackage{longtable}
\renewcommand{\arraystretch}{1.15} % slightly taller rows, so table rows read apart
\usepackage{pdfpages}
\usepackage{siunitx}
\sisetup{
  range-units = single,
  range-phrase = { to },
  per-mode = symbol,
}
\usepackage{xcolor}
\usepackage[most]{tcolorbox}

% Floats may fill more of a text page, so that no float page is left half empty under the page
% limit.

\renewcommand{\textfraction}{0.1}

\setlength{\floatsep}{10pt plus 2pt minus 2pt}

% TikZ diagrams (04-implementation/figures/*.tex) share font, palette and styles.
\usepackage{tikz}
\input{diagram-styles}

% --- Page style ----------------------------------------------------------------------------
\usepackage{fancyhdr}
\setlength{\headheight}{15pt}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\nouppercase{\rightmark}}
\fancyhead[R]{\thepage}
\renewcommand{\headrulewidth}{0.4pt}
% The head names the first chapter on the page (\rightmark is the first mark of the page).
\renewcommand{\chaptermark}[1]{\markboth{\thechapter\quad #1}{\thechapter\quad #1}}
\renewcommand{\sectionmark}[1]{}

% --- Bibliography --------------------------------------------------------------------------
\usepackage[backend=biber, style=ieee, sorting=nyt]{biblatex}
\addbibresource{bibliography.bib}

% --- Cross references (load last) ----------------------------------------------------------
\usepackage[hidelinks]{hyperref}
\usepackage[capitalise, noabbrev]{cleveref}

\NewDocumentCommand\vecsym{m}{\symbf{#1}}
