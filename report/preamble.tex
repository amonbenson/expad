% --- Semantic commands for content ---------------------------------------------------------

\NewDocumentCommand\vecsym{m}{\symbf{#1}}

% --- Scaffold commands (remove together with every use once the report is written) ----------

% What a paragraph has to convey.
\newtcolorbox{scaffoldbox}{
  breakable, enhanced,
  colback=scaffold!5, colframe=scaffold!70, boxrule=0.4pt, arc=1pt,
  left=4pt, right=4pt, top=3pt, bottom=3pt,
  fontupper=\small\sffamily\color{scaffold!90!black},
}
\NewDocumentCommand\scaffold{+m}{\begin{scaffoldbox}#1\end{scaffoldbox}}

% Scaffold text inside captions, table cells and other places where a box does not fit.
\NewDocumentCommand\scaffoldinline{m}{{\sffamily\color{scaffold}[#1]}}

% A figure that does not exist yet: what it shows and how to obtain it.
\NewDocumentCommand\placeholderfigure{O{0.9\linewidth} +m}{%
  \begin{tcolorbox}[
    width=#1, enhanced, center,
    colback=black!4, colframe=black!45, boxrule=0.4pt, sharp corners,
    borderline={0.4pt}{2pt}{black!45, dashed},
    title={Figure to be created}, coltitle=black, colbacktitle=black!12,
    fonttitle=\small\sffamily\bfseries, fontupper=\small\sffamily,
  ]#2\end{tcolorbox}%
}

% Information or a reason that is missing and has to come from the author. The first argument
% is the short form for the list of open questions.
\newcounter{openquestion}
\crefname{openquestion}{open question}{open questions}
\DeclareNewTOC[listname={List of Open Questions}]{loq}
\NewDocumentCommand\openquestion{m +m}{%
  \refstepcounter{openquestion}%
  \addcontentsline{loq}{figure}{\protect\numberline{\theopenquestion}#1}%
  \begin{tcolorbox}[
    breakable, enhanced,
    colback=openquestion!5, colframe=openquestion!75, boxrule=0.6pt, arc=1pt,
    left=4pt, right=4pt, top=3pt, bottom=3pt,
    title={Open question \theopenquestion: #1}, fonttitle=\small\sffamily\bfseries,
    fontupper=\small\sffamily,
  ]#2\end{tcolorbox}%
}
